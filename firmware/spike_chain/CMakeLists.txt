add_executable(spike_chain main.c)
target_include_directories(spike_chain PRIVATE ${CMAKE_CURRENT_LIST_DIR}/.. ${REPO_ROOT}/src)
target_compile_definitions(spike_chain PRIVATE PICO_STACK_SIZE=0x18000)
pico_set_linker_script_var(spike_chain RAM_LENGTH 320k)
pico_set_linker_script_var(spike_chain SCRATCH_Y_ORIGIN 0x20050000)
pico_set_linker_script_var(spike_chain SCRATCH_Y_LENGTH 96k)
target_link_libraries(spike_chain pico_stdlib hardware_watchdog)
pico_enable_stdio_usb(spike_chain 1)
pico_enable_stdio_uart(spike_chain 0)
pico_embed_pt_in_binary(spike_chain ${CMAKE_CURRENT_LIST_DIR}/../pt/pt.json)
pico_set_uf2_family(spike_chain "absolute")
pico_add_extra_outputs(spike_chain)

pico_enable_stdio_uart(spike_chain_app 0)
